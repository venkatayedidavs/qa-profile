Why this lesson is here. The page's own order: offline tests before the hook is enabled.

Create: test/approval-hook.test.cjs. These tests launch only the supplied script with synthetic stdin. They do not invoke editor tools, an AI service, or the commands described in an event.

const test = require('node:test');
const assert = require('node:assert/strict');
const { spawnSync } = require('node:child_process');
const path = require('node:path');

function sampleEvent(harness, toolName = 'example_read') {
  const args = { note: 'synthetic-canary-not-a-secret' };
  return harness === 'copilot'
    ? { toolName, toolArgs: args }
    : { hook_event_name: 'PreToolUse', tool_name: toolName,
        tool_input: args };
}

function invoke(harness, input) {
  return spawnSync(process.execPath, [
    path.join(__dirname, '../scripts/approval-hook.cjs'), harness,
  ], {
    input: typeof input === 'string' ? input : JSON.stringify(input),
    encoding: 'utf8',
    timeout: 2000,
  });
}

function readDecision(result, harness) {
  assert.equal(result.status, 0, result.stderr);
  assert.equal(result.stderr, '');
  assert.equal(result.stdout.includes('synthetic-canary'), false);
  const output = JSON.parse(result.stdout);
  if (harness === 'local') {
    assert.equal(output.hookSpecificOutput.hookEventName, 'PreToolUse');
    return output.hookSpecificOutput.permissionDecision;
  }
  return output.permissionDecision;
}

for (const harness of ['copilot', 'local']) {
  test(`${harness}: asks for a valid request`, () => {
    const result = invoke(harness, sampleEvent(harness));
    assert.equal(readDecision(result, harness), 'ask');
  });

  test(`${harness}: also asks for an unfamiliar tool`, () => {
    const result = invoke(harness, sampleEvent(harness, 'future_tool'));
    assert.equal(readDecision(result, harness), 'ask');
  });

  test(`${harness}: denies incomplete or malformed input`, () => {
    const incomplete = harness === 'copilot'
      ? { toolName: 'example_read' }
      : { hook_event_name: 'PreToolUse', tool_name: 'example_read' };
    const invalidInputs = [
      '', '{broken', '{}', 'null', '[]', 'true', incomplete,
      sampleEvent(harness, ''), sampleEvent(harness, '  '),
      sampleEvent(harness, 42),
    ];
    if (harness === 'local') {
      invalidInputs.push({
        ...sampleEvent(harness), hook_event_name: 'PostToolUse',
      });
      for (const toolInput of [null, [], 'text', 42, true]) {
        invalidInputs.push({
          ...sampleEvent(harness), tool_input: toolInput,
        });
      }
    }
    for (const input of invalidInputs) {
      const result = invoke(harness, input);
      assert.equal(readDecision(result, harness), 'deny');
    }
  });

  test(`${harness}: denies the other input format`, () => {
    const other = harness === 'local' ? 'copilot' : 'local';
    const result = invoke(harness, sampleEvent(other));
    assert.equal(readDecision(result, harness), 'deny');
  });
}

test('rejects an unsupported harness', () => {
  const result = invoke('unknown', sampleEvent('copilot'));
  assert.equal(result.status, 2);
  assert.equal(result.stdout, '');
});

Run node --test test/approval-hook.test.cjs: expect 9 passes, 0 failures. Then run node --test: the supplied files produce 18 tests; any cases you added increase that number.

Prepare, but do not activate yet: keep both configuration examples under examples/hooks/, where they are not workspace hook files. The Copilot version uses the shared SDK format, while Local expects its own event payload and output wrapper.

examples/hooks/copilot.json, for a Copilot session on Agent Host or the documented Copilot CLI runtime:

{
  "version": 1,
  "hooks": {
    "preToolUse": [
      {
        "type": "command",
        "bash": "node scripts/approval-hook.cjs copilot",
        "powershell": "node scripts/approval-hook.cjs copilot",
        "cwd": ".",
        "timeoutSec": 5
      }
    ]
  }
}

examples/hooks/local.json, only for the VS Code Local harness:

{
  "hooks": {
    "PreToolUse": [
      {
        "type": "command",
        "command": "node scripts/approval-hook.cjs local",
        "cwd": ".",
        "timeout": 5
      }
    ]
  }
}

Pass when: the offline contract is tested, including a deliberate regression. This still does not prove that the editor discovers or runs the hook.
