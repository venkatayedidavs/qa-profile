Step 16: A test-scoped fixture

Why this lesson is here. Repetition is refactored only once the cart tests exist. This fixture is the first shared setup.

Outcome: before use prepares; afterward cleans up. Each test gets a unique cart despite shared synthetic identity.

Study: fixtures. Add wrappers only for real repetition.

src/fixtures/shop.ts

import { test as base, expect, type APIRequestContext } from '@playwright/test';
import { cartSchema } from '../schemas/shop.js';

type CartFixture = { client: APIRequestContext; id: string };

export const test = base.extend<{ cart: CartFixture }>({
  cart: async ({ playwright, baseURL }, use) => {
    if (!baseURL) throw new Error('The lab baseURL must be configured');
    const client = await playwright.request.newContext({
      baseURL, extraHTTPHeaders: { Authorization: 'Bearer learner-alex' },
    });
    let cartId: string | undefined;
    try {
      const created = await client.post('/api/carts');
      expect(created.status()).toBe(201);
      cartId = cartSchema.parse(await created.json()).id;
      await use({ client, id: cartId });
    } finally {
      try {
        if (cartId) {
          const removed = await client.delete(`/api/carts/${cartId}`);
          expect([204, 404]).toContain(removed.status());
        }
      } finally {
        await client.dispose();
      }
    }
  },
});

export { expect } from '@playwright/test';

Pass when: same assertions pass independently in either order.
