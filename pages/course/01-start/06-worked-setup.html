Why this lesson is here. These are the server and config files Step 2 tells you to create. The original page supplies them before the steps begin.

Use these six supplied files in Step 2. You do not need to write a server before learning your first test. Use Node.js 22 LTS, version 22.18 or later, Git, and VS Code. The same Node major runs in CI; tsx executes TypeScript while a separate compiler checks types.

lab/server.ts. Loopback only; each start resets carts. Logs include method, path, status, and request ID, not bodies or authorization headers.

import express, { type Request, type Response, type NextFunction } from 'express';
import { randomUUID } from 'node:crypto';
import { z } from 'zod';

const app = express();
const products = [
  { id: 'p1', name: 'Notebook', priceCents: 1200, inStock: true },
  { id: 'p2', name: 'Pen', priceCents: 300, inStock: true },
  { id: 'p3', name: 'Bag', priceCents: 4900, inStock: false },
];
type Item = { productId: string; quantity: number; unitPriceCents: number };
type Cart = { id: string; owner: string; items: Item[] };
const carts = new Map<string, Cart>();
const accounts = new Map([
  ['Bearer learner-alex', 'alex'],
  ['Bearer learner-sam', 'sam'],
]);
const quantitySchema = z.object({ quantity: z.number().int().min(1).max(5) });

app.disable('x-powered-by');
app.use((request, response, next) => {
  const requestId = randomUUID();
  response.setHeader('x-request-id', requestId);
  response.once('finish', () => {
    console.log(JSON.stringify({ requestId, method: request.method,
      path: request.path, status: response.statusCode }));
  });
  next();
});
app.use(express.json({ limit: '10kb' }));

function cartView(cart: Cart) {
  const totalCents = cart.items.reduce(
    (sum, item) => sum + item.unitPriceCents * item.quantity, 0,
  );
  return { ...cart, totalCents };
}

function ownedCart(request: Request, response: Response): Cart | undefined {
  const owner = accounts.get(request.get('authorization') ?? '');
  if (!owner) {
    response.status(401).json({ error: 'Authentication required' });
    return;
  }
  const cart = carts.get(String(request.params.cartId));
  if (!cart) {
    response.status(404).json({ error: 'Cart not found' });
    return;
  }
  if (cart.owner !== owner) {
    response.status(403).json({ error: 'Access denied' });
    return;
  }
  return cart;
}

app.get('/health', (_request, response) => {
  response.json({ status: 'ready' });
});
app.get('/api/products', (request, response) => {
  const query = String(request.query.q ?? '').trim().toLowerCase();
  const matches = products.filter((product) =>
    product.name.toLowerCase().includes(query));
  response.json({ products: matches });
});
app.post('/api/carts', (request, response) => {
  const owner = accounts.get(request.get('authorization') ?? '');
  if (!owner) {
    response.status(401).json({ error: 'Authentication required' });
    return;
  }
  const cart: Cart = { id: randomUUID(), owner, items: [] };
  carts.set(cart.id, cart);
  response.status(201).json(cartView(cart));
});
app.get('/api/carts/:cartId', (request, response) => {
  const cart = ownedCart(request, response);
  if (cart) response.json(cartView(cart));
});
app.put('/api/carts/:cartId/items/:productId', (request, response) => {
  const cart = ownedCart(request, response);
  if (!cart) return;
  const payload = quantitySchema.safeParse(request.body);
  if (!payload.success) {
    response.status(400).json({ error: 'Quantity must be an integer from 1 to 5' });
    return;
  }
  const product = products.find((candidate) => candidate.id === request.params.productId);
  if (!product) {
    response.status(404).json({ error: 'Product not found' });
    return;
  }
  if (!product.inStock) {
    response.status(409).json({ error: 'Product out of stock' });
    return;
  }
  cart.items = cart.items.filter((item) => item.productId !== product.id);
  cart.items.push({ productId: product.id, quantity: payload.data.quantity,
    unitPriceCents: product.priceCents });
  response.json(cartView(cart));
});
app.delete('/api/carts/:cartId', (request, response) => {
  const cart = ownedCart(request, response);
  if (!cart) return;
  carts.delete(cart.id);
  response.status(204).end();
});
app.use(express.static('lab/public'));
app.use((error: unknown, _request: Request, response: Response, _next: NextFunction) => {
  response.status(error instanceof SyntaxError ? 400 : 500)
    .json({ error: 'Request could not be processed' });
});
app.listen(3100, '127.0.0.1', () => {
  console.log('Practice Shop: http://127.0.0.1:3100');
});

lab/public/app.js

const productList = document.querySelector('#products');
const catalogStatus = document.querySelector('#catalog-status');
const catalogError = document.querySelector('#catalog-error');
const searchButton = document.querySelector('#search-button');

async function loadProducts(query) {
  searchButton.disabled = true;
  productList.replaceChildren();
  catalogError.textContent = '';
  catalogStatus.textContent = 'Loading products...';
  try {
    const response = await fetch(`/api/products?q=${encodeURIComponent(query)}`);
    if (!response.ok) throw new Error('Catalog request failed');
    const body = await response.json();
    for (const product of body.products) {
      const item = document.createElement('li');
      const heading = document.createElement('h3');
      heading.textContent = product.name;
      const price = document.createElement('p');
      price.textContent = `SEK ${(product.priceCents / 100).toFixed(2)}`;
      const stock = document.createElement('p');
      stock.textContent = product.inStock ? 'In stock' : 'Out of stock';
      item.append(heading, price, stock);
      productList.append(item);
    }
    catalogStatus.textContent = `${body.products.length} products found`;
  } catch {
    catalogStatus.textContent = '';
    catalogError.textContent = 'Products unavailable. Try again.';
  } finally {
    searchButton.disabled = false;
  }
}

document.querySelector('#search-form').addEventListener('submit', (event) => {
  event.preventDefault();
  void loadProducts(document.querySelector('#search').value);
});
document.querySelector('#cart-form').addEventListener('submit', async (event) => {
  event.preventDefault();
  const status = document.querySelector('#cart-status');
  status.textContent = 'Loading cart...';
  try {
    const cartId = document.querySelector('#cart-id').value;
    const response = await fetch(`/api/carts/${encodeURIComponent(cartId)}`, {
      headers: { Authorization: 'Bearer learner-alex' },
    });
    if (!response.ok) throw new Error('Cart request failed');
    const cart = await response.json();
    status.textContent = `Cart total: SEK ${(cart.totalCents / 100).toFixed(2)}`;
  } catch {
    status.textContent = 'Cart unavailable.';
  }
});
